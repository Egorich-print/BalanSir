# BalanSir Raspberry Pi 3B+ (64-bit) boot configuration.
# Debug UART: serial console on the GPIO UART (pins 8/10) @115200.
#
# miniuart-bt moves UART0 (PL011) to GPIO 14/15 and hands Bluetooth the mini
# UART; enable_uart=1 forces a fixed core clock (needed for a stable baud on
# the mini UART path) and is harmless here. The kernel console uses the
# portable `serial0` alias (firmware maps it to the primary UART = PL011).

start_file=start.elf
fixup_file=fixup.dat

kernel=Image

# Disable overscan
disable_overscan=1

gpu_mem_256=100
gpu_mem_512=100
gpu_mem_1024=100

# Debug UART: PL011 on GPIO 14/15 for the serial console (recovery/logs).
enable_uart=1
dtoverlay=miniuart-bt

# 64-bit support
arm_64bit=1
